<script>
(function wait() {
  if (document.body.dataset.ready !== '1' || !window.__meadow) { setTimeout(wait, 50); return; }
  const m = window.__meadow, r6 = v => Math.round(v * 1e6) / 1e6;
  const fl = () => m.flowers.map(f => [f.id, f.kind, r6(f.x), r6(f.y), r6(f.R), r6(f.rot), f.seed, f.n, r6(f.f0), f.leaves.length, f.sub ? f.sub.length : 0, r6(f.ox), r6(f.oy), r6(f.age), f.clicked ? 1 : 0, f.fading ? r6(f.fade) : -1]);
  const air = () => { const o = []; for (const [x, y] of [[100, 100], [800, 500], [1500, 900], [400, 700]]) { m.windAt(x, y, m.t); o.push([x, y, r6(m.wind.x), r6(m.wind.y), r6(m.wind.ax), r6(m.wind.ay), r6(m.wind.px), r6(m.wind.py), r6(m.wind.g)]); } return o; };
  m.flies.length = 0;
  // The opening moment nodded and booked flowers for its butterflies: undone, so only the air moves them.
  for (const f of m.flowers) { f.vx = 0; f.vy = 0; f.ox = 0; f.oy = 0; f.busy = 0; f.lift = 0; f.dimT = 0; }
  const out = { W: m.W, H: m.H, S: m.S, planted: fl() };
  for (let i = 0; i < 60; i++) m.step(1 / 60);
  out.t60 = { air: air(), flowers: fl() };
  // pointer stir: set ptr and step, as the input would at 60 Hz
  const p = m.ptr; p.has = true; p.x = 300; p.y = 300; p.sx = 300; p.sy = 300; p.ax = 300; p.ay = 300;
  for (let i = 0; i < 30; i++) { p.x = 300 + i * 12; p.y = 300 + i * 5; m.step(1 / 60); }
  out.stir = { t: m.t, air: air(), samples: [[640, 420], [500, 380], [700, 450]].map(([x, y]) => { m.windAt(x, y, m.t); return [x, y, r6(m.wind.px), r6(m.wind.py)]; }) };
  p.has = false;
  // clicks: open ground and a bloom
  const f0 = m.flowers[m.flowers.length - 1];
  m.clickAt(f0.x, f0.y);
  out.clickPoints = [[f0.x, f0.y]];
  for (const [x, y] of [[0.04, 0.07], [0.04, 0.07], [0.75, 0.16], [0.31, 0.9], [0.9, 0.57], [0.16, 0.44], [0.62, 0.11]].map(([u, v]) => [Math.round(u * m.W), Math.round(v * m.H)])) { m.clickAt(x, y); out.clickPoints.push([x, y]); m.step(1 / 60); }
  out.clicks = { flowers: fl(), dust: m.dust.length };
  for (let i = 0; i < 120; i++) m.step(1 / 60);
  out.t180 = { t: m.t, air: air(), flowers: fl() };
  const pre = document.createElement('pre'); pre.id = 'out'; pre.textContent = JSON.stringify(out); document.body.appendChild(pre);
})();
</script>
