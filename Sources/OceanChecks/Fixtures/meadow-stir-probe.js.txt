<script>
// The m1 stir probe (fix round 2): the sketch's own pointer handlers and `stirStep`, fed per-step positions, the
// pointer-stir field recorded. Appended to a scratch copy of the sketch before `</body>`, with one hook added before
// its `if (FIXTURE)` line: `window.__run = () => { paused = false; };` (opened with `freeze=1`, so no frame loop runs
// and the probe steps by hand; `__run` lifts the pause so the move handler stirs as it does live).
(function wait() {
  if (document.body.dataset.ready !== '1' || !window.__meadow || !window.__run) { setTimeout(wait, 50); return; }
  const m = window.__meadow, r6 = v => Math.round(v * 1e6) / 1e6;
  const canvas = document.querySelector('canvas');
  window.__run();
  m.flies.length = 0;   // no butterflies: no downwash, nothing landing on the pointer
  const send = (type, x, y) => canvas.dispatchEvent(new PointerEvent(type, { clientX: x, clientY: y, pointerId: 1,
    isPrimary: true, pointerType: 'mouse', bubbles: true }));
  // A scenario is a list of ops: ['m', x, y] a pointermove, ['l'] a pointerleave, ['s'] one step of 1/60 s,
  // ['t', v] the Slow motion setting, ['snap'] the field now (its non-zero samples, [index, px, py]).
  const path = t => [300 + 500 * t + 200 * Math.sin(4 * t), 450 + 250 * Math.sin(2.5 * t)];
  const ops = {};
  const at = (list, k, every, per) => { for (let j = per - 1; j >= 0; j--) { const t = (k - j / per) / 60; if (k % every === 0) list.push(['m', ...path(t).map(r6)]); } };
  const run = (name, steps, every, per, extra) => {
    const o = []; o.push(['m', ...path(0).map(r6)]);
    for (let k = 1; k <= steps; k++) { if (extra) extra(o, k); at(o, k, every, per); o.push(['s']); if (k % 15 === 0) o.push(['snap']); }
    o.push(['l']); ops[name] = o;
  };
  run('rate60', 90, 1, 1);
  run('rate30', 90, 2, 1);
  run('rate120', 90, 1, 2);
  // Fast: 150 pt a step (9000 pt/s, past the ±1500 clamp, several stirs a step), then still.
  { const o = [['m', 200, 300]]; for (let k = 1; k <= 40; k++) { if (k <= 20) o.push(['m', 200 + 150 * k, 300 + 20 * k]); o.push(['s']); if (k % 10 === 0) o.push(['snap']); } o.push(['l']); ops.fast = o; }
  // Leave after a step, come back far off, go on.
  { const o = [['m', 250, 250]]; for (let k = 1; k <= 30; k++) { o.push(['m', 250 + 14 * k, 250 + 6 * k]); o.push(['s']); if (k % 10 === 0) o.push(['snap']); }
    o.push(['l']); for (let k = 0; k < 10; k++) o.push(['s']); o.push(['snap']);
    o.push(['m', 1300, 750]); for (let k = 1; k <= 30; k++) { o.push(['m', 1300 - 12 * k, 750 - 9 * k]); o.push(['s']); if (k % 10 === 0) o.push(['snap']); }
    o.push(['l']); ops.reentry = o; }
  // Slow motion 0.4: a step is 1/24 s of the pointer's real time.
  { const o = [['t', 0.4], ['m', 400, 400]]; for (let k = 1; k <= 40; k++) { o.push(['m', 400 + 8 * k, 400 - 3 * k]); o.push(['s']); if (k % 10 === 0) o.push(['snap']); }
    o.push(['l'], ['t', 1]); ops.slow = o; }
  const out = { W: m.W, H: m.H, S: m.S, columns: Math.ceil(m.W / (48 * m.S)) + 1, length: m.PFX.length, scenarios: {} };
  for (const name of Object.keys(ops)) {
    m.PFX.fill(0); m.PFY.fill(0);
    const snaps = [];
    for (const op of ops[name]) {
      if (op[0] === 'm') send('pointermove', op[1], op[2]);
      else if (op[0] === 'l') send('pointerleave', 0, 0);
      else if (op[0] === 's') m.step(1 / 60);
      else if (op[0] === 't') m.set.timeScale = op[1];
      else if (op[0] === 'snap') { const c = []; for (let i = 0; i < m.PFX.length; i++) if (m.PFX[i] !== 0 || m.PFY[i] !== 0) c.push([i, r6(m.PFX[i]), r6(m.PFY[i])]); snaps.push(c); }
    }
    out.scenarios[name] = { ops: ops[name], snaps };
  }
  const pre = document.createElement('pre'); pre.id = 'out'; pre.textContent = JSON.stringify(out); document.body.appendChild(pre);
})();
</script>
